cmake_minimum_required(VERSION 3.25)
project(pcsx5_boundary_fixture LANGUAGES CXX)

include("${GUARD_FILE}")
set(core "${CMAKE_CURRENT_BINARY_DIR}/fixture-core")
file(MAKE_DIRECTORY "${core}/include/pcsx5/core" "${core}/src")
file(WRITE "${core}/include/pcsx5/core/value.h" "#pragma once\n#include <cstdint>\n")
file(WRITE "${core}/src/value.cpp" "#include <pcsx5/core/value.h>\nint value() { return 1; }\n")
if(CASE STREQUAL "local_flags")
    add_subdirectory(local_flags)
else()
    add_library(fixture_core STATIC "${core}/src/value.cpp")
    target_include_directories(fixture_core PUBLIC "${core}/include")
endif()

if(CASE STREQUAL "positive" OR CASE STREQUAL "post_config_edit")
    # Comments do not create dependencies, and a literal local header is legal.
    file(APPEND "${core}/src/value.cpp"
        "// #include <windows.h>\n/* #include <vulkan/vulkan.h> */\n#include \"../include/pcsx5/core/value.h\"\n#include <limits>\n#include <optional>\nconstexpr int separated = 1'000;\nconstexpr char apostrophe = '\\'';\nconstexpr auto marker = \"/* quoted comment marker */\";\n")
elseif(CASE STREQUAL "windows")
    file(APPEND "${core}/src/value.cpp" "#include <windows.h>\n")
elseif(CASE STREQUAL "posix")
    file(APPEND "${core}/src/value.cpp" "#include <sys/mman.h>\n")
elseif(CASE STREQUAL "ui")
    file(APPEND "${core}/src/value.cpp" "#include <QWidget>\n")
elseif(CASE STREQUAL "vulkan")
    file(APPEND "${core}/src/value.cpp" "#include <vulkan/vulkan.h>\n")
elseif(CASE STREQUAL "metal")
    file(APPEND "${core}/src/value.cpp" "#include <Metal/Metal.h>\n")
elseif(CASE STREQUAL "legacy")
    file(APPEND "${core}/src/value.cpp" "#include <src/kernel/kernel.h>\n")
elseif(CASE STREQUAL "nested")
    file(APPEND "${core}/include/pcsx5/core/value.h" "#include <windows.h>\n")
elseif(CASE STREQUAL "macro")
    file(APPEND "${core}/src/value.cpp" "#define HOST_HEADER <windows.h>\n#include HOST_HEADER\n")
elseif(CASE STREQUAL "splice")
    file(APPEND "${core}/src/value.cpp" "#inc\\\nlude <windows.h>\n")
elseif(CASE STREQUAL "alternate")
    file(APPEND "${core}/src/value.cpp" "%:include <windows.h>\n")
elseif(CASE STREQUAL "import")
    file(APPEND "${core}/src/value.cpp" "import <windows.h>;\n")
elseif(CASE STREQUAL "include_next")
    file(APPEND "${core}/src/value.cpp" "#include_next <vector>\n")
elseif(CASE STREQUAL "escape")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/outside.h" "#pragma once\n")
    file(APPEND "${core}/src/value.cpp" "#include \"../../outside.h\"\n")
elseif(CASE STREQUAL "link")
    target_link_libraries(fixture_core PRIVATE legacy_kernel)
elseif(CASE STREQUAL "false_link")
    target_link_libraries(fixture_core PUBLIC OFF)
elseif(CASE STREQUAL "quoted_comment")
    file(APPEND "${core}/src/value.cpp" "const char* a = \"/*\";\n#include <unistd.h>\nconst char* b = \"*/\";\n")
elseif(CASE STREQUAL "pragma_link")
    file(APPEND "${core}/src/value.cpp" "#pragma comment(lib, \"legacy_kernel\")\n")
elseif(CASE STREQUAL "tight_import")
    file(APPEND "${core}/src/value.cpp" "#import<unistd.h>\n")
elseif(CASE STREQUAL "tight_include_next")
    file(APPEND "${core}/src/value.cpp" "#include_next<vector>\n")
elseif(CASE STREQUAL "cmake_header")
    file(WRITE "${core}/CMakeLists.txt" "#include <windows.h>\n")
    file(APPEND "${core}/src/value.cpp" "#include \"../CMakeLists.txt\"\n")
elseif(CASE STREQUAL "control_whitespace")
    string(ASCII 12 form_feed)
    file(APPEND "${core}/src/value.cpp" "#${form_feed}include <windows.h>\n")
elseif(CASE STREQUAL "transitive_link")
    add_library(host_wrapper INTERFACE)
    target_link_libraries(host_wrapper INTERFACE legacy_kernel)
    target_link_libraries(fixture_core PRIVATE host_wrapper)
elseif(CASE STREQUAL "interface_link")
    target_link_libraries(fixture_core INTERFACE legacy_kernel)
elseif(CASE STREQUAL "external_source")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/outside.cpp" "int outside() { return 2; }\n")
    target_sources(fixture_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/outside.cpp")
elseif(CASE STREQUAL "include_root")
    target_include_directories(fixture_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
elseif(CASE STREQUAL "inherited_include")
    include_directories("${CMAKE_CURRENT_BINARY_DIR}")
elseif(CASE STREQUAL "generated")
    set_source_files_properties("${core}/src/value.cpp" PROPERTIES GENERATED TRUE)
elseif(CASE STREQUAL "genex_source")
    target_sources(fixture_core PRIVATE "$<$<CONFIG:Debug>:${core}/src/value.cpp>")
elseif(CASE STREQUAL "genex_include")
    target_include_directories(fixture_core PRIVATE "$<BUILD_INTERFACE:${core}/include>")
elseif(CASE STREQUAL "compile_option")
    target_compile_options(fixture_core PRIVATE -include outside.h)
elseif(CASE STREQUAL "source_option")
    set_source_files_properties("${core}/src/value.cpp" PROPERTIES COMPILE_OPTIONS "-include;outside.h")
elseif(CASE STREQUAL "directory_definition")
    add_compile_definitions(HOST_HEADER=1)
elseif(CASE STREQUAL "global_flags")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -include outside.h")
elseif(CASE STREQUAL "local_flags")
    # Applied in the child directory which owns fixture_core.
elseif(CASE STREQUAL "pch")
    target_precompile_headers(fixture_core PRIVATE windows.h)
elseif(CASE STREQUAL "late_link")
    cmake_language(DEFER CALL target_link_libraries fixture_core PRIVATE legacy_kernel)
else()
    message(FATAL_ERROR "Unknown fixture case: ${CASE}")
endif()

cmake_language(DEFER CALL pcsx5_check_core_target fixture_core "${core}")
get_filename_component(guard_directory "${GUARD_FILE}" DIRECTORY)
add_custom_target(fixture_boundary_check
    COMMAND "${CMAKE_COMMAND}" "-DCORE_ROOT=${core}"
        -P "${guard_directory}/CheckCoreIncludes.cmake" VERBATIM)
add_dependencies(fixture_core fixture_boundary_check)
