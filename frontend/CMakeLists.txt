add_library(pcsx5_frontend STATIC src/session.cpp)
target_compile_features(pcsx5_frontend PUBLIC cxx_std_23)
target_link_libraries(pcsx5_frontend PUBLIC pcsx5_execution pcsx5_core)
target_include_directories(pcsx5_frontend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
pcsx5_enable_warnings(pcsx5_frontend)
if(ANDROID)
    add_library(pcsx5_app SHARED android/jni.cpp)
    target_link_libraries(pcsx5_app PRIVATE pcsx5_frontend pcsx5_runtime_posix)
    pcsx5_enable_warnings(pcsx5_app)
else()
    add_executable(pcsx5 src/main.cpp)
    target_link_libraries(pcsx5 PRIVATE pcsx5_frontend)
    if(APPLE)
        target_link_libraries(pcsx5 PRIVATE pcsx5_runtime_posix)
    elseif(WIN32)
        target_link_libraries(pcsx5 PRIVATE pcsx5_runtime_windows)
    else()
        target_link_libraries(pcsx5 PRIVATE pcsx5_runtime_linux)
    endif()
    pcsx5_enable_warnings(pcsx5)
    install(TARGETS pcsx5 RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_test(NAME pcsx5_frontend_self_test COMMAND pcsx5 --self-test)
        add_test(NAME pcsx5_frontend_version COMMAND pcsx5 --version)
        add_test(NAME pcsx5_frontend_reject_argument COMMAND pcsx5 --invalid)
        set_tests_properties(pcsx5_frontend_self_test pcsx5_frontend_version
            pcsx5_frontend_reject_argument PROPERTIES TIMEOUT 30 LABELS frontend)
        set_tests_properties(pcsx5_frontend_reject_argument PROPERTIES WILL_FAIL TRUE)
    endif()
endif()
